cmake_minimum_required(VERSION 3.20)
project(GyroGolfNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenGL REQUIRED)
find_package(glfw3 CONFIG REQUIRED)
add_executable(gyro_golf_native src/main.cpp)
target_link_libraries(gyro_golf_native PRIVATE glfw OpenGL::GL)
target_include_directories(gyro_golf_native PRIVATE third_party)
